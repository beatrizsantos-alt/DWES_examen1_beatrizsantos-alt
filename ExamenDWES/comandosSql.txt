CREATE DATABASE CAE;

USE CAE;
CREATE TABLE SOLICITUD (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(50) NOT NULL,
    apellidos VARCHAR(100) NOT NULL,
    dni VARCHAR(9) NOT NULL,
    f_nac DATE NOT NULL,
    tlf VARCHAR(15) NOT NULL,
    email VARCHAR(100) NOT NULL,
    profesion VARCHAR(30) NOT NULL CHECK (
        profesion IN (
            'soldadura',
            'informatica',
            'asistencia sociosanitaria'
        )
    ),
    jornadaParcial BOOLEAN NOT NULL CHECK (jornadaParcial IN (0, 1)),
    idiomas VARCHAR(30)
);
